After Effects Elastic Bounce (reference)

// position — boundary snap with 10% squash feel
vel = 0;
amp = .08;
freq = 2.2;
decay = 6;
n = 0;
if (numKeys > 0){
  n = nearestKey(time).index;
  if (key(n).time > time) n--;
}
if (n > 0){
  t = time - key(n).time;
  if (t < 0.35){
    v = velocityAtTime(key(n).time - .001);
    return value + v*amp*Math.sin(freq*t*2*Math.PI)/Math.exp(decay*t);
  }
}
value;
